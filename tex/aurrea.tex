% --- Adibide-ingurunea: zenbakitze globala ------------------------------------
% Quarto crossref-ek \begin{example} ingurunea sortzen du amsthm-ekin,
% kapituluko zenbakitzearekin ("Adibidea 4.1"). Hizkuntzalaritzako
% konbentzioa liburu osoko zenbakitzea da: (1), (2), (3)... crossref-ek bere
% div-ak quarto-pre fasean "custom AST" nodo bihurtzen ditu erabiltzailearen
% iragazkientzat ikusegin, beraz konponbidea ingurunea bera birdefinitzea da.
% Inguruneak crossref-ek utzitako ~ lausoa neutralizatzen du, eta
% \everypar bidez lehenengo benetako paragrafoaren hasieran jartzen du
% (n) etiketa, marjinan zintzilik: horrela "(1) a. ..." layout estandarra
% lortzen da, bai zerrendadun adibideentzat, bai perpaus bakarrekoentzat,
% bai glossadunentzat. \label/\ref ondo funtzionatzen dute.
% Ariketak (#exr-) eta definizioak (#def-) kapituluko zenbakitzearekin
% geratzen dira, nahi bezala.
\makeatletter
\newcounter{adibkont}
\newif\ifadib@jarrita
\def\adib@etiketa{%
  \ifadib@jarrita\else
    \adib@jarritatrue
    \llap{\makebox[2.4em][r]{(\arabic{adibkont})}\enspace}%
  \fi
}
\def\adib@empty{}
\AtBeginDocument{%
  \renewenvironment{example}[1][]{%
    \par\addvspace{1.2ex}%
    \refstepcounter{adibkont}%
    \adib@jarritafalse
    \let~\adib@empty
    \edef\adib@savedeverypar{\the\everypar}%
    \everypar{\adib@etiketa\everypar{\adib@savedeverypar}}%
    % Zerrenda-adibideak: LaTeX-en \everypar mekanismoa listek baliogabetzen
    % dute, eta lehenengo \item-aren aurretik. Soluzioa: lehenengo itemaren
    % etiketan sartu zenbakia -> "(1) a. ..." layout estandarra.
    \let\adib@oldatitem\@item
    \def\@item[##1]{%
      \ifadib@jarrita
        \adib@oldatitem[{##1}]%
      \else
        \adib@jarritatrue
        \adib@oldatitem[{\makebox[2.4em][r]{(\arabic{adibkont})}\enspace##1}]%
      \fi
    }%
    \ignorespaces
  }{%
    \everypar{\adib@savedeverypar}%
    \par\addvspace{1.2ex}%
  }
  % Ariketak: kapituluko zenbakitzea, baina marjinan "(4.1)" gisa (HTML-ekin
  % bateratzeko); amsthm-aren "Ariketa 4.1" goiburua ordezten du.
  \renewenvironment{exercise}[1][]{%
    \par\addvspace{1.2ex}\refstepcounter{exercise}%
    \noindent\llap{\makebox[2.4em][r]{(\theexercise)}\enspace}\ignorespaces
  }{%
    \par\addvspace{1.2ex}%
  }%
  % Zuhaitzak adibideen barruan (#exm- div batean .tree div sartuta):
  % forest-ek ez du \everypar abiarazten, bertikal moduan boxea eraikitzen
  % duelako, eta beraz (n) etiketa ez litzateke inprimatuko. \forest
  % (ingurune-hasierako makroa) kiribatzen dugu: etiketa inprimatu behar
  % bada, \noindent egin (horrek \everypar abiarazi eta etiketa errendatzen
  % du, edo etiketa jada jarrita badago, ezer ez), eta gero forest abiarazi.
  % Ezkonduta dagoen \ifadib@jarrita etiketa bikoiztea ekiditen du.
  \let\adib@oldforest\forest
  \def\forest{%
    \noindent\adib@etiketa
    \adib@oldforest
  }%
}
\makeatother
